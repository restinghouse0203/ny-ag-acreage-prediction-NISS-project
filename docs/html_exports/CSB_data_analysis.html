2. Geometry Exploration¶

The dataset contains a 'geometry' column (likely in WKT format). We convert this text into geometric objects using shapely and create a GeoDataFrame for spatial operations.

# Check if 'geometry' column exists
if 'geometry' in df.columns:
    # Examine first entry to detect format: WKT (str), WKB (bytes) or bytes-as-string
    first_geom = df['geometry'].iloc[0]

    # Prepare a column of shapely geometries named 'geom'
    from shapely.wkb import loads as wkb_loads
    from shapely.wkt import loads as wkt_loads
    import ast
    from shapely.geometry.base import BaseGeometry

    def parse_geom(val):
        # bytes (WKB)
        if isinstance(val, (bytes, bytearray)):
            return wkb_loads(val)
        # string: try WKT first
        if isinstance(val, str):
            # common case: literal b'...' stored as string
            if val.startswith("b'") or val.startswith('b"'):
                try:
                    b = ast.literal_eval(val)
                    return wkb_loads(b)
                except Exception:
                    pass
            # try WKT
            try:
                return wkt_loads(val)
            except Exception:
                # try interpreting as utf-8 bytes then WKB
                try:
                    b = val.encode('utf-8')
                    return wkb_loads(b)
                except Exception:
                    raise
        # If it's already a shapely geometry, return as-is
        if isinstance(val, BaseGeometry):
            return val
        # Unknown format
        raise TypeError('Unsupported geometry format')

    # Apply parsing with safe error handling for large datasets
    print('Parsing geometry column to shapely objects (this may take some time)...')
    try:
        df['geom'] = df['geometry'].apply(lambda v: parse_geom(v))
    except Exception as e:
        # Attempt vectorized fallback: try using GeoPandas from_wkb if available
        print(f'Error parsing geometries with custom parser: {e}')
        try:
            # geopandas can read WKB via GeoSeries.from_wkb (pandas array of bytes)
            gpd_series = gpd.GeoSeries.from_wkb(df['geometry'])
            df['geom'] = gpd_series
        except Exception as e2:
            print(f'Fallback parsing also failed: {e2}')
            raise

    # Create a GeoDataFrame
    try:
        gdf = gpd.GeoDataFrame(df, geometry='geom')
    except Exception as e:
        print(f'Failed to create GeoDataFrame: {e}')
        raise

    print('GeoDataFrame created.')

    # Plot the spatial data (sample to avoid performance issues)
    print('Plotting sample of data...')
    sample_size = min(10000, len(gdf))
    
    # Plot customization for visibility:
    # - Increase marker size or use edge color for polygons
    # - Use alpha=1.0 for maximum visibility of thin lines
    # - Coordinates are projected (likely meters), not Lat/Lon
    gdf.sample(sample_size).plot(figsize=(10, 10), alpha=1.0, edgecolor='blue', linewidth=0.3)
    plt.title(f"Spatial Distribution (Sample of {sample_size})\nProjected Coordinates (Meters)")
    plt.xlabel('Easting (m)')
    plt.ylabel('Northing (m)')
    plt.show()

# 1. Group & Classify Crop Sequences
if cdl_cols:
    print("Classifying crop sequences...")
    
    # Get sequences as list of codes
    sequences = df[cdl_cols].values
    
    def classify_rotation(seq):
        unique_crops = set(seq)
        unique_names = [get_short_name(c) for c in unique_crops]
        
        # 1. Continuous (Single crop dominant)
        # Allow 1-2 years of "noise" or slight deviation in long series if needed, but strictly:
        if len(unique_crops) == 1:
            return f"Cont. {unique_names[0]}"
        
        # 2. Simple Rotation (Two crops)
        if len(unique_crops) == 2:
            crops_str = "-".join(sorted(unique_names))
            return f"Rot {crops_str}"
            
        # 3. Three Crop Rotation
        if len(unique_crops) == 3:
             # Prioritize common ones like Corn-Soy-Wht
             crops_str = "-".join(sorted(unique_names))
             return f"Rot {crops_str}"
        
        # 4. Complex
        return "Complex/Mixed"

    # Apply classification
    df['Rotation_Type'] = [classify_rotation(row) for row in sequences]
    
    # Count rotation types
    rotation_counts = df['Rotation_Type'].value_counts().head(10)
    
    print("\nTop 10 Crop Rotations (Grouped Categories):")
    rot_df = rotation_counts.reset_index()
    rot_df.columns = ['Rotation Type', 'Count']
    display(rot_df)
    
    plt.figure(figsize=(10, 5))
    sns.barplot(data=rot_df, x='Rotation Type', y='Count', palette='viridis', hue='Rotation Type', legend=False)

# 2. Time Series Analysis
if cdl_cols:
    # Melt to long format
    melted = df[cdl_cols].melt(var_name='Year_Col', value_name='Crop_Code')
    melted['Year'] = melted['Year_Col'].str.replace('CDL', '').astype(int)
    
    # Map codes to FULL names for plot legend
    melted['Crop_Name'] = melted['Crop_Code'].map(cdl_full_mapping).fillna(melted['Crop_Code'].astype(str))
    
    # Filter for top 5 crops
    top_crops_names = melted['Crop_Name'].value_counts().head(5).index
    filtered_melted = melted[melted['Crop_Name'].isin(top_crops_names)]
    
    plt.figure(figsize=(12, 6))
    sns.countplot(data=filtered_melted, x='Year', hue='Crop_Name', palette='Set2')
    plt.title("Distribution of Top 5 Crop Types Over Time")
    plt.ylabel("Number of Fields")
    plt.xlabel("Year")
    plt.legend(title='Crop Name')
    plt.grid(axis='y', linestyle='--', alpha=0.7)
    plt.show()
    
    # --- Statistical Metrics (ACF, Unit Root) ---
    print("\n--- Time Series Statistical Metrics ---")
    
    # Aggregate counts per year for each major crop to create time series
    ts_data = melted.groupby(['Year', 'Crop_Name']).size().unstack(fill_value=0)
    
    stats_results = []
    
    for crop in top_crops_names:
        series = ts_data[crop]
        
        # 1. Autocorrelation (ACF) at lag 1
        acf_val = acf(series, nlags=1)[1] if len(series) > 1 else np.nan
        
        # 2. ADF Unit Root Test
        try:
            adf_result = adfuller(series)
            adf_stat = adf_result[0]
            p_value = adf_result[1]
            is_stationary = p_value < 0.05
        except Exception:
            adf_stat = np.nan
            p_value = np.nan
            is_stationary = "Error"
            
        stats_results.append({
            'Crop': crop,
            'ACF (Lag 1)': round(acf_val, 3),
            'ADF Statistic': round(adf_stat, 3),
            'P-Value': round(p_value, 4),
            'Stationary?': is_stationary
        })
    
    stats_df = pd.DataFrame(stats_results)
    display(stats_df)
    print("Note: High P-Value (>0.05) implies non-stationarity (trend/instability).")
